  const handleChange = (event) => {
    const { name, value, type, checked } = event.target;

    setForm((previous) => ({
      ...previous,
      [name]: type === "checkbox" ? checked : value,
    }));

    setErrors((previous) => ({
      ...previous,
      [name]: "",
      form: "",
    }));

    setSuccessMessage("");
  };

  const validateForm = () => {
    const newErrors = {};

    if (!form.email.trim()) {
      newErrors.email = "Please enter your email address.";
    } else if (
      !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(form.email.trim())
    ) {
      newErrors.email = "Please enter a valid email address.";
    }

    if (!form.password) {
      newErrors.password = "Please enter your password.";
    }

    return newErrors;
  };

    event.preventDefault();

    setSuccessMessage("");

    const validationErrors = validateForm();
    setErrors(validationErrors);

      "Your details are valid. Login will be available when authentication is connected to the Kampusly server."
    );

function getStudentProfile() {
let profile = {};

try {
const savedProfile = localStorage.getItem("kampuslyProfile");


if (savedProfile) {
  profile = JSON.parse(savedProfile);
}


} catch (error) {
console.error("Could not load Kampusly profile:", error);
}

return {
name:
typeof profile.name === "string" && profile.name.trim()
? profile.name.trim()
: localStorage.getItem("kampuslyName") || "Student",
avatar: localStorage.getItem("kampuslyAvatar") || "",
};
}

function getInitials(name) {
return (
name
.trim()
.split(/\s+/)
.filter(Boolean)
.slice(0, 2)
.map((part) => part[0])
.join("")
.toUpperCase() || "S"
);
}

function formatName(fullName) {
return fullName.trim().split(/\s+/)[0] || "Student";
}

function StudentAvatar({
name,
avatar,
className = "",
imageClassName = "",
}) {
const [imageFailed, setImageFailed] = useState(false);

useEffect(() => {
setImageFailed(false);
}, [avatar]);

if (avatar && !imageFailed) {
return (
<img
src={avatar}
alt={`${name}'s profile`}
className={`${className} ${imageClassName}`.trim()}
onError={() => setImageFailed(true)}
/>
);
}

return (
<span
className={className}
role="img"
aria-label={`${name}'s initials`}
>
{getInitials(name)} </span>
);
}

import { useState } from "react";
import { UserPlus, UserCheck, LoaderCircle } from "lucide-react";

export default function FollowButton({
  userId,
  initialIsFollowing = false,
  onFollowChange,
}) {
  const [isFollowing, setIsFollowing] = useState(initialIsFollowing);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState("");

  const handleFollow = async () => {
    if (!userId || loading) return;

    setLoading(true);
    setError("");

    try {
      const token = localStorage.getItem("token");

      const response = await fetch(
        `http://localhost:5001/api/users/${userId}/follow`,
        {
          method: isFollowing ? "DELETE" : "POST",
          headers: {
            "Content-Type": "application/json",
            ...(token ? { Authorization: `Bearer ${token}` } : {}),
          },
        }
      );

      const data = await response.json();

      if (!response.ok) {
        throw new Error(data.message || "Unable to update follow status.");
      }

      const nextFollowing =
        typeof data.isFollowing === "boolean"
          ? data.isFollowing
          : !isFollowing;

      setIsFollowing(nextFollowing);

      if (onFollowChange) {
        onFollowChange(nextFollowing, data);
      }
    } catch (err) {
      setError(err.message || "Something went wrong.");
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="kampusly-follow-wrapper">
      <button
        type="button"
        className={`kampusly-follow-button ${
          isFollowing ? "is-following" : ""
        }`}
        onClick={handleFollow}
        disabled={loading}
        aria-label={isFollowing ? "Unfollow student" : "Follow student"}
      >
        {loading ? (
          <LoaderCircle size={16} className="follow-spinner" />
        ) : isFollowing ? (
          <UserCheck size={16} />
        ) : (
          <UserPlus size={16} />
        )}

        <span>
          {loading ? "Please wait..." : isFollowing ? "Following" : "Follow"}
        </span>
      </button>

      {error && (
        <p className="kampusly-follow-error" role="alert">
          {error}
        </p>
      )}
    </div>
  );
}

    setNotice(null);
  };

  const saveProfile = (event) => {
    event.preventDefault();

    if (!draft.name.trim()) {
      showNotice("Please enter your full name.", "error");
      return;
    }

    if (
      draft.email.trim() &&
      !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(draft.email.trim())
    ) {
      showNotice("Please enter a valid email address.", "error");
      return;
    }

    if (draft.website.trim()) {
      try {
        const parsedWebsite = new URL(draft.website.trim());

        if (!["http:", "https:"].includes(parsedWebsite.protocol)) {
          showNotice("Please enter a valid website link.", "error");
          return;
        }
      } catch {
        showNotice(
          "Enter a complete website link, including https://.",
          "error"
        );
        return;
      }
    }

    const updatedProfile = {
      ...draft,
      name: draft.name.trim(),
      username: draft.username.trim().replace(/^@/, ""),
      bio: draft.bio.trim(),
      department: draft.department.trim(),
      faculty: draft.faculty.trim(),
      matricNumber: draft.matricNumber.trim(),
      email: draft.email.trim(),
      location: draft.location.trim(),
      website: draft.website.trim(),
      interests: Array.isArray(draft.interests) ? draft.interests : [],
    };

        "kampuslyProfile",
        JSON.stringify(updatedProfile)

      localStorage.setItem("kampuslyEmail", updatedProfile.email);

      setDraft(updatedProfile);
      setEditMode(false);
      showNotice("Your profile has been updated successfully.");
    } catch (error) {
      console.error("Could not save profile:", error);
      showNotice(
        "Unable to save your profile. Your browser storage may be full.",
        "error"

      setIsSaving(false);

      showNotice("Please select a valid image file.", "error");
      event.target.value = "";
      return;
    }

    if (file.size > 2 * 1024 * 1024) {
      showNotice("Your image must be smaller than 2 MB.", "error");
      event.target.value = "";
      return;
    }

    reader.onload = () => {

          return;

const [feedback, setFeedback] = useState("");

const items = useMemo(() => {
if (!category) return [];


const search = searchTerm.trim().toLowerCase();

let results = category.examples.filter((item) => {
  const matchesSearch =
    item.title.toLowerCase().includes(search) ||
    item.description.toLowerCase().includes(search) ||
    item.seller.toLowerCase().includes(search);

  const matchesFilter =
    activeFilter === "All" ||
    activeFilter === `All ${categoryId.replace("-", " ")}` ||
    item.title.toLowerCase().includes(activeFilter.toLowerCase()) ||
    item.description.toLowerCase().includes(activeFilter.toLowerCase());

  return matchesSearch && matchesFilter;
});

if (sortBy === "price-low") {
  results = [...results].sort(
    (a, b) => (a.price ?? Infinity) - (b.price ?? Infinity)
  );
}

if (sortBy === "price-high") {
  results = [...results].sort(
    (a, b) => (b.price ?? -Infinity) - (a.price ?? -Infinity)
  );
}

return results;


}, [category, categoryId, searchTerm, activeFilter, sortBy]);

if (!category) {
return ( <div className="service-category-not-found"> <Package size={42} /> <h1>Service category not found</h1> <p>Choose a category from the Kampusly Services page.</p>
<button type="button" onClick={() => navigate("/student/services")}>

);

const CategoryIcon = category.icon;

const toggleSaved = (itemId) => {
setSavedItems((current) =>
current.includes(itemId)
? current.filter((id) => id !== itemId)
: [...current, itemId]
);


setFeedback(
  savedItems.includes(itemId) ? "Removed from saved items." : "Item saved."
);


};

const contactSeller = (item) => {
setFeedback(`Contacting ${item.seller} will be available when messaging is connected.`);
};

const createPost = () => {
navigate("/student/feed", {
state: {
serviceCategory: category.title,
openCreatePost: true,
},
});
};

return ( <div className="service-category-page"> <header className="service-category-header">

                    onChange={(event) => setDescription(event.target.value)}
                  />

                  <p className="withdraw-field-hint">
                    {description.length}/160 characters
                  </p>
                </div>

                <div className="withdraw-info-box">
                  <Clock3 size={19} />
                  <div>
                    <strong>Processing time</strong>
                    <p>
                      Actual processing times will depend on the payment
                      provider and bank once withdrawals are connected.
                    </p>
                  </div>
                </div>

                <button
                  type="submit"
                  className="withdraw-primary-button"
                  disabled={
                    !bankAccounts.some(
                      (account) =>
                        account.id === selectedBank && account.accountNumber
                    )
                  }
                >
                  Review withdrawal
                  <ChevronRight size={19} />
                </button>

                <p className="withdraw-secure-note">
                  <LockKeyhole size={15} />
                  Demo only â€” no money will be moved.
                </p>
              </form>
            </section>

            <aside className="withdraw-side-card">
              <h3>Before you withdraw</h3>

              <div className="withdraw-tip">
                <span>01</span>
                <p>Make sure your bank account details are correct.</p>
              </div>

              <div className="withdraw-tip">
                <span>02</span>
                <p>Check your available balance before requesting a withdrawal.</p>
              </div>

              <div className="withdraw-tip">
                <span>03</span>
                <p>Review the withdrawal details before confirming.</p>
              </div>

              <div className="withdraw-side-divider" />

              <div className="withdraw-fee-row">
                <span>Withdrawal fee</span>
                <strong>Not configured</strong>
              </div>

              <div className="withdraw-fee-row">
                <span>Minimum amount</span>
                <strong>â‚¦1,000</strong>
              </div>

              <div className="withdraw-side-note">
                <Info size={17} />
                <p>
                  Fees and limits must be confirmed when the real wallet
                  backend is implemented.
                </p>
              </div>
            </aside>
          </div>
        )}
      </div>

      {showReview && (
        <div
          className="withdraw-modal-backdrop"
          onMouseDown={(event) => {
            if (event.target === event.currentTarget) {
              setShowReview(false);
            }
          }}
        >
          <section
            className="withdraw-review-modal"
            role="dialog"
            aria-modal="true"
            aria-labelledby="withdraw-review-title"
          >
            <button
              type="button"
              className="withdraw-modal-close"
              onClick={() => setShowReview(false)}
              aria-label="Close review"
            >
              <X size={21} />
            </button>

            <div className="withdraw-modal-icon">
              <ArrowDownLeft size={25} />
            </div>

            <h2 id="withdraw-review-title">Review withdrawal</h2>
            <p className="withdraw-modal-subtitle">
              Check these details before continuing.
            </p>

            <div className="withdraw-review-amount">
              <span>Withdrawal amount</span>
              <strong>{formatMoney(numericAmount)}</strong>
            </div>

            <div className="withdraw-review-row">
              <span>Destination bank</span>
              <strong>{selectedAccount?.bankName}</strong>
            </div>

            <div className="withdraw-review-row">
              <span>Account name</span>
              <strong>{selectedAccount?.accountName}</strong>
            </div>

            <div className="withdraw-review-row">
              <span>Account number</span>
              <strong>{selectedAccount?.accountNumber}</strong>
            </div>

            <div className="withdraw-review-row">
              <span>Description</span>
              <strong>{description.trim() || "Not provided"}</strong>
            </div>

            <div className="withdraw-demo-warning">
              This is a UI demonstration. Confirming will not transfer money
              or create a real withdrawal request.
            </div>

            <button
              type="button"
              className="withdraw-primary-button"
              onClick={handleConfirmWithdrawal}
            >
              Confirm demo request
            </button>

            <button
              type="button"
              className="withdraw-cancel-button"
              onClick={() => setShowReview(false)}
            >
              Go back and edit
            </button>
          </section>
        </div>
      )}
    </main>
  );
